`TGeneralPolygon2`방식을 분석한다.
`TGeneralPolygon2`은 내부에 Outer/Hole로 이루어져있다. 우리 프로젝트같은 경우`PolygonsDifference`의 결과로 나온 `TGeneralPolygon2`은
Outer는 CCW로 인덱싱이 되어있고, Hole은 CW로 인덱싱이 되어있어서 구멍 Polygon과 구멍 밖 Polygon을 구별할 수 있다.

```cpp
bool Triangulate(const TGeneralPolygon2<RealType>& GeneralPolygon, TArray<FIndex3i>* TrianglesOut = nullptr, TArray<TVector2<RealType>>* VerticesOut = nullptr, bool bFallbackToGeneralizedWinding = false)
{
	TArray<TVector2<RealType>> AllVertices;
	TArray<FIndex2i> AllEdges;

	// 1. Outer와 Hole의 정점들을 모두 모은다.
	auto AppendVertices = [&AllVertices, &AllEdges](const TArray<TVector2<RealType>>& Vertices)
	{
		if (Vertices.IsEmpty())
		{
			return;
		}

		int32 StartIdx = AllVertices.Num();
		AllVertices.Append(Vertices);
		AllEdges.Reserve(AllEdges.Num() + Vertices.Num() - 1);
		int32 NumVertices = Vertices.Num();
		for (int32 Last = NumVertices - 1, Idx = 0; Idx < NumVertices; Last = Idx++)
		{
			AllEdges.Add(FIndex2i(StartIdx + Last, StartIdx + Idx));
		}
	};
	AppendVertices(GeneralPolygon.GetOuter().GetVertices());
	for (int32 HoleIdx = 0; HoleIdx < GeneralPolygon.GetHoles().Num(); HoleIdx++)
	{
		AppendVertices(GeneralPolygon.GetHoles()[HoleIdx].GetVertices());
	}
	// 1. end
	
	// 2. 들로네 삼각분할을 수행한다.	
	bool bSuccess = Triangulate(AllVertices, AllEdges);
	if (TrianglesOut)
	{
		// Outer의 Winding방향을 확인합니다. Hole은 Outer와 반대로 처리되어있다고 간주하고 없애야 하기 때문입니다.
		EFillMode FillMode = GeneralPolygon.OuterIsClockwise() ? EFillMode::NegativeWinding : EFillMode::PositiveWinding;
		if (bSuccess || !bFallbackToGeneralizedWinding)
		{
			// 여기서 어떤삼각형을 유지하고 제거할지 결정하게된다.
			// 결과 대입
			*TrianglesOut = GetFilledTriangles(AllEdges, FillMode);
		}
		else
		{
			// 삼각분할 실패시 fallback
			// winding 판별이 어려운 불완전한 경계에서 winding하는 함수라고한다 (상대적으로 느리다고 함)
			GetFilledTrianglesGeneralizedWinding(*TrianglesOut, AllVertices, AllEdges, FillMode);
		}
	}
	if (VerticesOut)
	{
		// 결과 대입
		*VerticesOut = MoveTemp(AllVertices);
	}
	return bSuccess;
}
```

```cpp
	// 전달받은 에지의 방향에 따라 Winding 변화량을 반환하는 함수
	// 처음 정렬: 입력받은 경계 에지를 `EdgeDelta`의 Key로 저장하기 위해 정렬.
    // 지금 정렬: 삼각분할 결과를 탐색하면서 전달받은 `Edge`를 `EdgeDelta`의 Key와 비교하기 위해 정렬.
	auto GetWindingChange = [&EdgeDelta](FIndex2i Edge)
	{
		int32 Winding = 0;
		int32 Sign = -1;
		if (Edge.A > Edge.B)
		{
			Sign = 1;
			Swap(Edge.A, Edge.B); // `EdgeDelta.Find(Edge)`로 값을 찾기 위해 정렬
		}
		int32* Change = EdgeDelta.Find(Edge);
		if (!Change)
		{
			return 0;
		}
		else
		{
			return Sign * (*Change);
		}
	};
	
	// 다음에 탐색할 삼각형의 정보를 `ToWalk에 저장하는 함수`
	auto WalkEdge = [FillMode, &EdgeDelta, &ToWalk, &GetWindingChange](const FIndex2i& Edge, int32 BeforeWinding)
	{
		if (FillMode == FDelaunay2::EFillMode::Solid)
		{
			// Solid일때는 정렬이 되어있지 않은 원본 Edge를 정렬시킨다.
			FIndex2i Unoriented = Edge;
			if (Unoriented.A > Unoriented.B)
			{
				Swap(Unoriented.A, Unoriented.B);
			}
			// 위에서 EdgeDelta에 저장하지 않았다면 : Winding을 1로 기록해서 저장한다.
			// 아마 삼각분할되서 생성된 삼각형들에 대해서 처리될 것이다.
			if (!EdgeDelta.Contains(Unoriented)) // true : 경계 Edge였다. false : 삼각분할로 생성된 일반 내부 Edge이기 때문에 반대편 삼각형을 탐색하도록 등록
			{
				ToWalk.Add({ FIndex2i(Edge.B, Edge.A), 1 });
			}
		}
		else
		{
			// Solid가 아니라 Winding방식이면 아니면 여기서 처리된다.
			// GetWindingChange : 
			// 경계 Edge가 아니면 => 0
			// 정렬된 경계 Edge면 => -1 * EdgeDelta.value
			// 정렬이 안된 경계 Edge면 => EdgeDelta.value가 retun된다.
			/*
				// Outer 경계 원본 에지: (3, 0)
				// EdgeDelta[(0, 3)] = -1
				
				// GhostTriangle에서 시작
				WalkEdge(FIndex2i(0, 3), 0);
				
				// GetWindingChange((0, 3)) = (-1) * (-1) = +1
				// Winding = 0 + 1 = 1
				
				// ToWalk에 저장되는 결과: {(3, 0), 1}
				
				// (1, 2)일때도 마찬가지다. 2, 1로 들어가서 sign이 1이되고 EdgeDelta가 1을 가지고있기때문에 1을 반환해서 [(2,1), 1]
			*/
			int32 Winding = BeforeWinding + GetWindingChange(Edge);
			ToWalk.Add({ FIndex2i(Edge.B, Edge.A), Winding });
		}
	};
```

```cpp

	// 실제로 삼각형을 하나씩 방문하면서 Winding 값을 기록하는 과정
	int32 PickIdx = 0;
	while (ToWalk.Num())
	{
		// Traverse the walk list in something similar to a breadth-first traversal
		// Non-closed shapes in the input edges will give different results depending on visit order
		//  and depth-first traversal would be more sensitive to starting triangle in that case
		// 요약하면 넓이우 선탐색처럼 진행한다는 뜻
		PickIdx = (PickIdx + 1) % ToWalk.Num();
		const FWalk Walk = ToWalk[PickIdx];
		ToWalk.RemoveAtSwap(PickIdx, EAllowShrinking::No);
		int32 Vert = EdgeToVert[Walk.Edge];
		FIndex3i UniqueTri = AsUniqueTriangle(Walk.Edge, Vert);
		if (UniqueTri.A < 0) // it's a ghost
		{
			continue;
		}
		int32* Winding = Windings.Find(UniqueTri);
		if (Winding)
		{
			bWellDefinedResult = bWellDefinedResult && (*Winding == Walk.Winding);
			continue;
		}
		Windings.Add(UniqueTri, Walk.Winding);		
		WalkEdge(FIndex2i(Walk.Edge.B, Vert), Walk.Winding);
		WalkEdge(FIndex2i(Vert, Walk.Edge.A), Walk.Winding);
	}

	TrianglesOut.Reset();
	if (FillMode == FDelaunay2::EFillMode::Solid)
	{
		TrianglesOut.Reserve(EdgeToVert.Num() / 3 - GhostTriCount - Windings.Num());
		// everything we *didn't* walk over is included
		EnumerateTriangles([&Windings, &TrianglesOut](const FIndex2i& Edge, int32 Vertex) -> bool
			{
				FIndex3i Triangle(Vertex, Edge.A, Edge.B);
				if (!Windings.Contains(Triangle))
				{
					TrianglesOut.Add(Triangle);
				}
				return true;
			}, true /*bSkipGhosts*/);
	}
	else
	{
		TrianglesOut.Reserve(Windings.Num());
		for (const TPair<FIndex3i, int32>& TriWinding : Windings)
		{
			if ((FillMode == FDelaunay2::EFillMode::NegativeWinding	&& TriWinding.Value < 0) ||
				(FillMode == FDelaunay2::EFillMode::NonZeroWinding	&& TriWinding.Value != 0) ||
				(FillMode == FDelaunay2::EFillMode::PositiveWinding && TriWinding.Value > 0) ||
				(FillMode == FDelaunay2::EFillMode::OddWinding		&& (TriWinding.Value % 2) != 0))
			{
				TrianglesOut.Add(TriWinding.Key);
			}
		}
	}
	return bWellDefinedResult;
}
```
