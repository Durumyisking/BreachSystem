우선 총알은 정밀한 충돌이 필요하다고 생각한다. (끄는 옵션을 줄 수는 있지만, 아무튼 파괴된 틈 사이로 쏘고, 이런 콘텐츠의 제작 가능성이 있다.)
그리고 애초에 벽처럼 보이는곳에 쐈는데 뚫리거나, 비주얼적으로 빈곳에 쐈는데 막힌다면 좋은 플레이경험이 아닐 것이다.

#### 이제 기획대로 구현해보자.

우선 DynamicMesh가 Simple과 Complex 둘 다 쓰도록 설정한다.

```cpp
DynamicMeshProxy->bEnableComplexCollision = true;
DynamicMeshProxy->CollisionType = CTF_UseSimpleAndComplex; // Simple, Complex둘 다 쓸거임!
```

메시 업데이트때 Bound기준으로 SimpleCollsion을 생성한다.
```cpp
bool UBreachableSurfaceComponent::UpdateMesh(UE::Geometry::FDynamicMesh3&& Mesh)
{
	if (!ensureMsgf(IsValid(DynamicMeshProxy), TEXT("%s requires a surface proxy to update its mesh."), *GetName()))
	{
		return false;
	}

	// 메시 기반으로 SimpleCollision을 생성한다. EX)박스형태
	FKAggregateGeom SimpleCollision;
	if (Mesh.TriangleCount() > 0)
	{
		// Mesh의 로컬 경계로 박스를 만든다.
		const auto Bounds = Mesh.GetBounds();
		const FVector3d Size = Bounds.Max - Bounds.Min;

		FKBoxElem Box;
		Box.Center = FVector((Bounds.Min + Bounds.Max) * 0.5);
		Box.X = Size.X;
		Box.Y = Size.Y;
		Box.Z = Size.Z;

		SimpleCollision.BoxElems.Add(Box);
	}
	DynamicMeshProxy->SetSimpleCollisionShapes(SimpleCollision, false);

	// 등록전에 메시를 갱신한다. (등록을 먼저하면 갱신 전 내용으로 충돌처리 할 수 있음)
	DynamicMeshProxy->SetMesh(MoveTemp(Mesh));
	DynamicMeshProxy->UpdateCollision();

	// 컴포넌트 등록 안되어있으면 
	const bool bFirstUpdate = !DynamicMeshProxy->IsRegistered();
	if (!DynamicMeshProxy->IsRegistered())
	{
		DynamicMeshProxy->RegisterComponent();
		SetCollisionEnabled(ECollisionEnabled::NoCollision);
		SetVisibility(false, false);
		DynamicMeshProxy->SetCollisionEnabled(OriginalCollisionEnabled);
		DynamicMeshProxy->SetVisibility(bOriginalVisible, false);
		DynamicMeshProxy->SetHiddenInGame(bOriginalHiddenInGame, false);
	}

	return true;
}
```
